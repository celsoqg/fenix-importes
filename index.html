// Código base — versão com melhorias pré‑Firebase const configApp = { modoTeste: true, usarLoginExterno: false, salvarDadosLocalmente: true }; let listaProdutos = []; function adicionarProduto(nome, precoCusto, margem) { if (!nome || precoCusto <= 0 || margem < 0) { return { ok: false, mensagem: "Dados inválidos, revise os valores" }; } const precoVenda = precoCusto * (1 + margem / 100); const novoProduto = { id: Date.now(), nome, precoCusto, margem, precoVenda: parseFloat(precoVenda.toFixed(2)) }; listaProdutos.push(novoProduto); if (configApp.salvarDadosLocalmente) { localStorage.setItem("produtos", JSON.stringify(listaProdutos)); } return { ok: true, produto: novoProduto }; } function carregarDadosLocais() { const dadosSalvos = localStorage.getItem("produtos"); if (dadosSalvos) listaProdutos = JSON.parse(dadosSalvos); } // Inicialização carregarDadosLocais(); console.log("Aplicativo carregado — modo teste ativado"); console.log("Lista de produtos atual:", listaProdutos);
