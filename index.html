async function uploadImageToGallery(event) { const file = event.target.files[0]; if (!file) return; const statusEl = document.getElementById('upload-status'); statusEl.classList.remove('hidden'); statusEl.innerText = '🤖 IA cortando fundo e otimizando imagem...'; statusEl.className = 'text-[10px] text-amber-500 font-bold animate-pulse mt-1'; // Pega a chave API do remove.bg salva nas configurações ou usa a chave fixa const REMOVE_BG_KEY = storeSettings.removeBgKey || 'SUA_CHAVE_AQUI'; if (REMOVE_BG_KEY && REMOVE_BG_KEY !== 'SUA_CHAVE_AQUI') { try { const formData = new FormData(); formData.append('image_file', file); formData.append('size', 'auto'); const response = await fetch('https://api.remove.bg/v1.0/removebg', { method: 'POST', headers: { 'X-Api-Key': REMOVE_BG_KEY }, body: formData }); if (response.ok) { const blob = await response.blob(); const reader = new FileReader(); reader.onloadend = function() { document.getElementById('prod-imageUrl').value = reader.result; statusEl.innerText = '✨ Fundo removido e imagem otimizada com sucesso!'; statusEl.className = 'text-[10px] text-green-400 font-bold mt-1'; }; reader.readAsDataURL(blob); return; } } catch (e) { console.warn("Falha no serviço externo, aplicando tratamento local.", e); } } // Tratamento local automático via Canvas (fallback) const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.onload = function() { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); canvas.width = 500; canvas.height = 500; const hRatio = canvas.width / img.width; const vRatio = canvas.height / img.height; const ratio = Math.min(hRatio, vRatio) * 0.85; const centerShift_x = (canvas.width - img.width * ratio) / 2; const centerShift_y = (canvas.height - img.height * ratio) / 2; ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.drawImage(img, 0, 0, img.width, img.height, centerShift_x, centerShift_y, img.width * ratio, img.height * ratio); document.getElementById('prod-imageUrl').value = canvas.toDataURL('image/png'); statusEl.innerText = '🖼️ Imagem enquadrada e pronta!'; statusEl.className = 'text-[10px] text-green-400 font-bold mt-1'; }; img.src = e.target.result; }; reader.readAsDataURL(file); }
